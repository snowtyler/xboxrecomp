cmake_minimum_required(VERSION 3.20)
project(nv2a_vsh_regression C)
enable_testing()
add_executable(nv2a-vsh-test nv2a_vsh_test.c ../../src/kernel/nv2a_vsh_interp.c)
target_include_directories(nv2a-vsh-test PRIVATE ../../src/kernel)
if(MSVC)
    target_compile_options(nv2a-vsh-test PRIVATE /UNDEBUG)
else()
    target_compile_options(nv2a-vsh-test PRIVATE -UNDEBUG)
endif()
if(NOT WIN32)
    target_link_libraries(nv2a-vsh-test PRIVATE m)
endif()
add_test(NAME nv2a-vsh-epilogue COMMAND nv2a-vsh-test)
add_subdirectory(../.. runtime)
add_executable(nv2a-v015-method-test method_test.c)
target_include_directories(nv2a-v015-method-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-v015-method-test PRIVATE xboxrecomp)
target_compile_options(nv2a-v015-method-test PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG>)
foreach(mode RANGE 0 1)
    add_test(NAME nv2a-v015-constant-methods-${mode} COMMAND nv2a-v015-method-test ${mode})
    set_tests_properties(nv2a-v015-constant-methods-${mode} PROPERTIES ENVIRONMENT
        "RECOMP_NV2A_V015_RENDER=${mode}")
endforeach()
