cmake_minimum_required(VERSION 3.20)
project(nv2a_clear_regression C)
enable_testing()
add_executable(nv2a-clear-test nv2a_clear_test.c)
target_include_directories(nv2a-clear-test PRIVATE ../../src/kernel)
if(MSVC)
    target_compile_options(nv2a-clear-test PRIVATE /UNDEBUG)
else()
    target_compile_options(nv2a-clear-test PRIVATE -UNDEBUG)
endif()
add_test(NAME nv2a-partial-clears COMMAND nv2a-clear-test)
add_executable(nv2a-clip-test nv2a_clip_test.c)
target_include_directories(nv2a-clip-test PRIVATE ../../src/kernel)
if(MSVC)
    target_compile_options(nv2a-clip-test PRIVATE /UNDEBUG)
else()
    target_compile_options(nv2a-clip-test PRIVATE -UNDEBUG)
    target_link_libraries(nv2a-clip-test PRIVATE m)
endif()
add_test(NAME nv2a-homogeneous-clipping COMMAND nv2a-clip-test)
