cmake_minimum_required(VERSION 3.20)
project(nv2a_gpu_device_probe C)
add_executable(nv2a-gpu-device-probe device_probe.c)
target_link_libraries(nv2a-gpu-device-probe PRIVATE d3d11 dxgi dxguid)
target_compile_options(nv2a-gpu-device-probe PRIVATE /UNDEBUG)
enable_testing()
add_test(NAME nv2a-hardware-device-readback COMMAND nv2a-gpu-device-probe)
add_executable(nv2a-gpu-combiner-test combiner_gpu_test.c
    ../../src/kernel/nv2a_combiner.c ../../src/kernel/nv2a_combiner_hlsl.c)
target_include_directories(nv2a-gpu-combiner-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-gpu-combiner-test PRIVATE d3d11 dxgi dxguid d3dcompiler)
target_compile_options(nv2a-gpu-combiner-test PRIVATE /UNDEBUG)
add_test(NAME nv2a-hardware-combiner-equivalence COMMAND nv2a-gpu-combiner-test)
add_executable(nv2a-gpu-raster-test raster_test.c ../../src/kernel/nv2a_hw_raster.c
    ../../src/kernel/nv2a_combiner.c ../../src/kernel/nv2a_combiner_hlsl.c ../../src/kernel/nv2a_vsh_hlsl.c)
target_include_directories(nv2a-gpu-raster-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-gpu-raster-test PRIVATE d3d11 dxgi dxguid d3dcompiler)
target_compile_options(nv2a-gpu-raster-test PRIVATE /UNDEBUG)
add_test(NAME nv2a-hardware-raster-coherence COMMAND nv2a-gpu-raster-test)
add_test(NAME nv2a-hardware-bulk-readback-coherence COMMAND nv2a-gpu-raster-test)
set_tests_properties(nv2a-hardware-bulk-readback-coherence PROPERTIES ENVIRONMENT "RECOMP_HW_BULK_READBACK=1")
add_test(NAME nv2a-hardware-texture-hash COMMAND nv2a-gpu-raster-test)
set_tests_properties(nv2a-hardware-texture-hash PROPERTIES ENVIRONMENT "RECOMP_HW_TEXTURE_HASH=1")
add_test(NAME nv2a-hardware-texture-epoch COMMAND nv2a-gpu-raster-test)
set_tests_properties(nv2a-hardware-texture-epoch PROPERTIES ENVIRONMENT "RECOMP_HW_TEXTURE_EPOCH=1")
add_test(NAME nv2a-hardware-texture-hash-epoch COMMAND nv2a-gpu-raster-test)
set_tests_properties(nv2a-hardware-texture-hash-epoch PROPERTIES ENVIRONMENT "RECOMP_HW_TEXTURE_HASH=1;RECOMP_HW_TEXTURE_EPOCH=1")
add_test(NAME nv2a-hardware-coalesced-inputs COMMAND nv2a-gpu-raster-test)
set_tests_properties(nv2a-hardware-coalesced-inputs PROPERTIES ENVIRONMENT "RECOMP_HW_RELEASE_COALESCE=1;RECOMP_HW_TEXTURE_HASH=1;RECOMP_HW_TEXTURE_EPOCH=1")
add_executable(nv2a-device-loss-test device_loss_test.c ../../src/kernel/nv2a_hw_raster.c
    ../../src/kernel/nv2a_combiner.c ../../src/kernel/nv2a_combiner_hlsl.c ../../src/kernel/nv2a_vsh_hlsl.c)
target_include_directories(nv2a-device-loss-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-device-loss-test PRIVATE d3d11 dxgi dxguid d3dcompiler)
target_compile_options(nv2a-device-loss-test PRIVATE /UNDEBUG)
target_compile_definitions(nv2a-device-loss-test PRIVATE NV2A_HW_TEST_DEVICE_LOSS)
add_test(NAME nv2a-hardware-device-removal COMMAND nv2a-device-loss-test)
add_executable(nv2a-gpu-vertex-test vertex_gpu_test.c ../../src/kernel/nv2a_vsh_interp.c
    ../../src/kernel/nv2a_vsh_hlsl.c)
target_include_directories(nv2a-gpu-vertex-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-gpu-vertex-test PRIVATE d3d11 dxgi dxguid d3dcompiler)
target_compile_options(nv2a-gpu-vertex-test PRIVATE /UNDEBUG)
add_test(NAME nv2a-hardware-vertex-equivalence COMMAND nv2a-gpu-vertex-test)
add_executable(nv2a-diagnostic-render-test diagnostic_render_test.c
    ../../src/kernel/nv2a_hw_raster.c ../../src/kernel/nv2a_combiner.c
    ../../src/kernel/nv2a_combiner_hlsl.c ../../src/kernel/nv2a_vsh_hlsl.c
    ../../src/kernel/nv2a_vsh_interp.c)
target_include_directories(nv2a-diagnostic-render-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-diagnostic-render-test PRIVATE d3d11 dxgi dxguid d3dcompiler)
target_compile_options(nv2a-diagnostic-render-test PRIVATE /UNDEBUG)
foreach(mode RANGE 0 1)
    add_test(NAME nv2a-v015-render-${mode} COMMAND nv2a-diagnostic-render-test)
    set_tests_properties(nv2a-v015-render-${mode} PROPERTIES ENVIRONMENT
        "RECOMP_NV2A_COLOR_CLAMP=${mode};RECOMP_NV2A_V015_RENDER=${mode}")
endforeach()
