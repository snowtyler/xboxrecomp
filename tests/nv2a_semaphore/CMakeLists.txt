cmake_minimum_required(VERSION 3.20)
project(nv2a_semaphore_test C)
add_subdirectory(../.. runtime)
add_executable(nv2a-semaphore-test semaphore_test.c)
target_include_directories(nv2a-semaphore-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-semaphore-test PRIVATE xboxrecomp)
if(MSVC)
    target_compile_options(nv2a-semaphore-test PRIVATE /UNDEBUG)
else()
    target_compile_options(nv2a-semaphore-test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME nv2a-semaphore-order COMMAND nv2a-semaphore-test ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/conformance/test.xbe)
add_test(NAME nv2a-semaphore-gpu-order COMMAND nv2a-semaphore-test ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/conformance/test.xbe --gpu)
