cmake_minimum_required(VERSION 3.20)
project(upstream_v014_integration C)
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
enable_testing()
foreach(name apu_irq apu_dsp_ack apu_lifecycle xaudio2)
    add_executable(${name}_integration ../${name}/test_main.c)
    target_include_directories(${name}_integration PRIVATE ../../src ../../src/apu ../../src/nv2a)
    target_link_libraries(${name}_integration PRIVATE xboxrecomp)
    target_compile_options(${name}_integration PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG>)
    add_test(NAME ${name} COMMAND ${name}_integration)
endforeach()
set_tests_properties(apu_dsp_ack PROPERTIES ENVIRONMENT "RECOMP_APU_DSP_ACK=auto")
add_executable(apu_period_integration ../apu_mixdown/monitor_test.c)
target_include_directories(apu_period_integration PRIVATE ../../src ../../src/apu)
target_link_libraries(apu_period_integration PRIVATE xboxrecomp)
add_test(NAME apu_period COMMAND apu_period_integration)
add_executable(port_integration port_test.c)
target_include_directories(port_integration PRIVATE ../../src/kernel)
target_link_libraries(port_integration PRIVATE xboxrecomp)
add_test(NAME guest_port_io COMMAND port_integration)
add_executable(apu_mixdown_integration ../apu_mixdown/test_main.c)
target_include_directories(apu_mixdown_integration PRIVATE ../../src ../../src/apu)
target_link_libraries(apu_mixdown_integration PRIVATE xboxrecomp)
add_test(NAME apu_mixdown_all COMMAND apu_mixdown_integration)
set_tests_properties(apu_mixdown_all PROPERTIES ENVIRONMENT
    "RECOMP_APU_MIXDOWN_ALL=1;APU_MIXDOWN_EXPECT=1")
add_test(NAME apu_mixdown_two_bins COMMAND apu_mixdown_integration)
set_tests_properties(apu_mixdown_two_bins PROPERTIES ENVIRONMENT
    "RECOMP_APU_MIXDOWN_ALL=0;APU_MIXDOWN_EXPECT=0")
