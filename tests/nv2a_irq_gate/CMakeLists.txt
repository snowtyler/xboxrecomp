cmake_minimum_required(VERSION 3.20)
project(nv2a_irq_gate_test C)
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
target_compile_definitions(xbox_kernel PRIVATE NV2A_IRQ_GATE_TEST)
add_executable(nv2a-irq-gate-test irq_gate_test.c)
target_include_directories(nv2a-irq-gate-test PRIVATE ../../src/kernel)
target_link_libraries(nv2a-irq-gate-test PRIVATE xboxrecomp)
if(MSVC)
 target_compile_options(nv2a-irq-gate-test PRIVATE /UNDEBUG)
endif()
enable_testing()
set(XBE ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/conformance/test.xbe)
add_test(NAME nv2a-irq-gate COMMAND nv2a-irq-gate-test ${XBE} on)
add_test(NAME nv2a-irq-control COMMAND nv2a-irq-gate-test ${XBE} off)
